/* ============================================================
   METRIC LAYOUT TABLE
   ============================================================ */

ARG Operational Metric Layout =
DATATABLE(
    "SortOrder", INTEGER,
    "MetricName", STRING,
    "MetricKey", STRING,
{
    {1, "CO Determinations (% Target)", "_coDeterminations"},
    {2, "CO TAT (Application Review)", "_coTatApplicationReview"},
    {3, "CO Avg Days Ahead of Target", "_coAvgDaysAheadOfTarget"},
    {4, "PA TAT (Application Review)", "_paTatApplicationReview"},
    {5, "PA Avg Days Ahead of Target", "_paAvgDaysAheadOfTarget"}
}
)

/* ============================================================
   TREND COLUMN VARIABLE
   ============================================================ */

VAR _trending = " "

/* ============================================================
   TIME PLACEHOLDERS
   ============================================================ */

VAR _year = [Current Fiscal Year]

VAR _priorYear = [Prior Fiscal YTD]

VAR _currentmonthsort = [Current Month Sort]

VAR _month3 = [Month Year Prior 2]

VAR _month2 = [Month Year Prior 1]

VAR _month1 = [Month Year Current]

VAR _displayMonth3 = [Month Display Prior 2]

VAR _displayMonth2 = [Month Display Prior 1]

VAR _displayMonth1 = [Month Display Current]

/* ============================================================
   OPERATIONAL EXCELLENCE PLACEHOLDERS
   ============================================================ */

VAR __coDeterminationsMonth3 = CALCULATE([Determinations], ALL(dimdate), dimdate[Month_Year] = _month3)
VAR __coDeterminationsMonth2 = CALCULATE([Determinations], ALL(dimdate), dimdate[Month_Year] = _month2)
VAR __coDeterminationsMonth1 = CALCULATE([Determinations], ALL(dimdate), dimdate[Month_Year] = _month1)
VAR __coDeterminationsYtd = CALCULATE([Determinations], ALL(dimdate), dimdate[Fiscal Year] = _year, dimdate[Month Sort] <= _currentmonthsort)
VAR __coDeterminationsTarget = CALCULATE([Determinations Target], ALL(dimdate), dimdate[Fiscal Year] = _year, dimdate[Month Sort] <= _currentmonthsort)
VAR __coDeterminationsDisplay = FORMAT(__coDeterminationsYtd, "#,##") & IF(ISBLANK(DIVIDE(__coDeterminationsYtd, __coDeterminationsTarget)), "", " (" & FORMAT(DIVIDE(__coDeterminationsYtd, __coDeterminationsTarget), "0%") & ")")

VAR __coTatApplicationReviewMonth3 = CALCULATE([CO Avg Processing Time], ALL(dimdate), dimdate[Month_Year] = _month3)
VAR __coTatApplicationReviewMonth2 = CALCULATE([CO Avg Processing Time], ALL(dimdate), dimdate[Month_Year] = _month2)
VAR __coTatApplicationReviewMonth1 = CALCULATE([CO Avg Processing Time], ALL(dimdate), dimdate[Month_Year] = _month1)

VAR __coAvgDaysAheadOfTargetMonth3 = CALCULATE([nan], ALL(dimdate), dimdate[Month_Year] = _month3)
VAR __coAvgDaysAheadOfTargetMonth2 = CALCULATE([nan], ALL(dimdate), dimdate[Month_Year] = _month2)
VAR __coAvgDaysAheadOfTargetMonth1 = CALCULATE([nan], ALL(dimdate), dimdate[Month_Year] = _month1)
VAR __coAvgDaysAheadOfTargetTargetMonth3 = CALCULATE([CO Avg Processing Time Target], ALL(dimdate), dimdate[Month_Year] = _month3)
VAR __coAvgDaysAheadOfTargetTargetMonth2 = CALCULATE([CO Avg Processing Time Target], ALL(dimdate), dimdate[Month_Year] = _month2)
VAR __coAvgDaysAheadOfTargetTargetMonth1 = CALCULATE([CO Avg Processing Time Target], ALL(dimdate), dimdate[Month_Year] = _month1)
VAR __coAvgDaysAheadOfTargetMonth3Display = FORMAT(DIVIDE(__coAvgDaysAheadOfTargetMonth3,100), "#,##0") & IF(ISBLANK(DIVIDE(__coAvgDaysAheadOfTargetMonth3, __coAvgDaysAheadOfTargetTargetMonth3)), "", " (" & FORMAT(DIVIDE(__coAvgDaysAheadOfTargetMonth3, __coAvgDaysAheadOfTargetTargetMonth3), "0%") & ")")
VAR __coAvgDaysAheadOfTargetMonth2Display = FORMAT(DIVIDE(__coAvgDaysAheadOfTargetMonth2,100), "#,##0") & IF(ISBLANK(DIVIDE(__coAvgDaysAheadOfTargetMonth2, __coAvgDaysAheadOfTargetTargetMonth2)), "", " (" & FORMAT(DIVIDE(__coAvgDaysAheadOfTargetMonth2, __coAvgDaysAheadOfTargetTargetMonth2), "0%") & ")")
VAR __coAvgDaysAheadOfTargetMonth1Display = FORMAT(DIVIDE(__coAvgDaysAheadOfTargetMonth1,100), "#,##0") & IF(ISBLANK(DIVIDE(__coAvgDaysAheadOfTargetMonth1, __coAvgDaysAheadOfTargetTargetMonth1)), "", " (" & FORMAT(DIVIDE(__coAvgDaysAheadOfTargetMonth1, __coAvgDaysAheadOfTargetTargetMonth1), "0%") & ")")

VAR __paTatApplicationReviewMonth3 = CALCULATE([PA Avg Processing Time], ALL(dimdate), dimdate[Month_Year] = _month3)
VAR __paTatApplicationReviewMonth2 = CALCULATE([PA Avg Processing Time], ALL(dimdate), dimdate[Month_Year] = _month2)
VAR __paTatApplicationReviewMonth1 = CALCULATE([PA Avg Processing Time], ALL(dimdate), dimdate[Month_Year] = _month1)

VAR __paAvgDaysAheadOfTargetMonth3 = CALCULATE([nan], ALL(dimdate), dimdate[Month_Year] = _month3)
VAR __paAvgDaysAheadOfTargetMonth2 = CALCULATE([nan], ALL(dimdate), dimdate[Month_Year] = _month2)
VAR __paAvgDaysAheadOfTargetMonth1 = CALCULATE([nan], ALL(dimdate), dimdate[Month_Year] = _month1)
VAR __paAvgDaysAheadOfTargetTargetMonth3 = CALCULATE([PA Avg Processing Time Target], ALL(dimdate), dimdate[Month_Year] = _month3)
VAR __paAvgDaysAheadOfTargetTargetMonth2 = CALCULATE([PA Avg Processing Time Target], ALL(dimdate), dimdate[Month_Year] = _month2)
VAR __paAvgDaysAheadOfTargetTargetMonth1 = CALCULATE([PA Avg Processing Time Target], ALL(dimdate), dimdate[Month_Year] = _month1)
VAR __paAvgDaysAheadOfTargetMonth3Display = FORMAT(DIVIDE(__paAvgDaysAheadOfTargetMonth3,100), "#,##0") & IF(ISBLANK(DIVIDE(__paAvgDaysAheadOfTargetMonth3, __paAvgDaysAheadOfTargetTargetMonth3)), "", " (" & FORMAT(DIVIDE(__paAvgDaysAheadOfTargetMonth3, __paAvgDaysAheadOfTargetTargetMonth3), "0%") & ")")
VAR __paAvgDaysAheadOfTargetMonth2Display = FORMAT(DIVIDE(__paAvgDaysAheadOfTargetMonth2,100), "#,##0") & IF(ISBLANK(DIVIDE(__paAvgDaysAheadOfTargetMonth2, __paAvgDaysAheadOfTargetTargetMonth2)), "", " (" & FORMAT(DIVIDE(__paAvgDaysAheadOfTargetMonth2, __paAvgDaysAheadOfTargetTargetMonth2), "0%") & ")")
VAR __paAvgDaysAheadOfTargetMonth1Display = FORMAT(DIVIDE(__paAvgDaysAheadOfTargetMonth1,100), "#,##0") & IF(ISBLANK(DIVIDE(__paAvgDaysAheadOfTargetMonth1, __paAvgDaysAheadOfTargetTargetMonth1)), "", " (" & FORMAT(DIVIDE(__paAvgDaysAheadOfTargetMonth1, __paAvgDaysAheadOfTargetTargetMonth1), "0%") & ")")

/* ============================================================
   OPERATIONAL EXCELLENCE STATUS
   ============================================================ */

VAR __coDeterminationsStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__coDeterminationsMonth1), BLANK(),
        __coDeterminationsMonth1 >= 10, "red",
        __coDeterminationsMonth1 >= 5, "amber",
        "green"
    )

VAR __coTatApplicationReviewStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__coTatApplicationReviewMonth1), BLANK(),
        __coTatApplicationReviewMonth1 >= 10, "red",
        __coTatApplicationReviewMonth1 >= 5, "amber",
        "green"
    )

VAR __coAvgDaysAheadOfTargetStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__coAvgDaysAheadOfTargetMonth1), BLANK(),
        __coAvgDaysAheadOfTargetMonth1 >= 10, "red",
        __coAvgDaysAheadOfTargetMonth1 >= 5, "amber",
        "green"
    )

VAR __paTatApplicationReviewStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__paTatApplicationReviewMonth1), BLANK(),
        __paTatApplicationReviewMonth1 >= 10, "red",
        __paTatApplicationReviewMonth1 >= 5, "amber",
        "green"
    )

VAR __paAvgDaysAheadOfTargetStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__paAvgDaysAheadOfTargetMonth1), BLANK(),
        __paAvgDaysAheadOfTargetMonth1 >= 10, "red",
        __paAvgDaysAheadOfTargetMonth1 >= 5, "amber",
        "green"
    )

/* ============================================================
   DYNAMIC HTML CREATION
   ============================================================ */
VAR _operationalRows =
    CONCATENATEX(
        'ARG Operational Metric Layout',
        VAR _metricKey = [MetricKey]

    VAR _status =
        SWITCH(
            _metricKey,
            "_coDeterminations", __coDeterminationsStatus,
            "_coTatApplicationReview", __coTatApplicationReviewStatus,
            "_coAvgDaysAheadOfTarget", __coAvgDaysAheadOfTargetStatus,
            "_paTatApplicationReview", __paTatApplicationReviewStatus,
            "_paAvgDaysAheadOfTarget", __paAvgDaysAheadOfTargetStatus
        )

    VAR _ytd =
        SWITCH(
            _metricKey,
            "_coDeterminations", __coDeterminationsDisplay
        )

    VAR _monthPrior2 =
        SWITCH(
            _metricKey,
            "_coDeterminations", __coDeterminationsMonth3,
            "_coTatApplicationReview", __coTatApplicationReviewMonth3,
            "_coAvgDaysAheadOfTarget", __coAvgDaysAheadOfTargetMonth3Display,
            "_paTatApplicationReview", __paTatApplicationReviewMonth3,
            "_paAvgDaysAheadOfTarget", __paAvgDaysAheadOfTargetMonth3Display
        )

    VAR _monthPrior1 =
        SWITCH(
            _metricKey,
            "_coDeterminations", __coDeterminationsMonth2,
            "_coTatApplicationReview", __coTatApplicationReviewMonth2,
            "_coAvgDaysAheadOfTarget", __coAvgDaysAheadOfTargetMonth2Display,
            "_paTatApplicationReview", __paTatApplicationReviewMonth2,
            "_paAvgDaysAheadOfTarget", __paAvgDaysAheadOfTargetMonth2Display
        )

    VAR _currentMonth =
        SWITCH(
            _metricKey,
            "_coDeterminations", __coDeterminationsMonth1,
            "_coTatApplicationReview", __coTatApplicationReviewMonth1,
            "_coAvgDaysAheadOfTarget", __coAvgDaysAheadOfTargetMonth1Display,
            "_paTatApplicationReview", __paTatApplicationReviewMonth1,
            "_paAvgDaysAheadOfTarget", __paAvgDaysAheadOfTargetMonth1Display
        )





        VAR _trendHtml = ""


    VAR _formatString =
        SWITCH(
            _metricKey,
            "_coDeterminations", "#,##0",
            "_coTatApplicationReview", "#,##0",
            "_coAvgDaysAheadOfTarget", "#,##0",
            "_paTatApplicationReview", "#,##0",
            "_paAvgDaysAheadOfTarget", "#,##0"
        )

        VAR _ytdHtml =
            IF(
                ISBLANK(_ytd),
                "",
                    FORMAT(_ytd, _formatString)
            )
        VAR _monthPrior2Html =
            IF(
                ISBLANK(_monthPrior2),
                "",
                    FORMAT(_monthPrior2, _formatString)
            )
        VAR _monthPrior1Html =
            IF(
                ISBLANK(_monthPrior1),
                "",
                    FORMAT(_monthPrior1, _formatString)
            )
        VAR _currentMonthHtml =
            IF(
                ISBLANK(_currentMonth),
                "",
                    FORMAT(_currentMonth, _formatString)
            )
        RETURN
            "<tr>"
                & "<td class='label'>" & [MetricName] & "</td>"
                & "<td class='status'><span class='dot "
                & COALESCE(_status, "") & "'></span></td>"
                & "<td class='number'>" & _ytdHtml & "</td>"
                & "<td class='number'>" & _monthPrior2Html & "</td>"
                & "<td class='number'>" & _monthPrior1Html & "</td>"
                & "<td class='number'>" & _currentMonthHtml & "</td>"
                & "<td class='number'>" & _trendHtml & "</td>"
                & "</tr>",
        "",
        [SortOrder],
        ASC
    )

RETURN
    "
    <!-- OPERATIONAL EXCELLENCE -->
    <tr>
      <td colspan='6' class='header'>
        Operational Excellence
      </td>
    </tr>
    <tr>
      <td class='left'>
        <table class='metric'>
          <colgroup>
            <col style='width:34%'>
            <col style='width:3%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
          </colgroup>
          <tr class='subheader'>
            <th></th>
            <th></th>
            <th>PY '" & FORMAT(RIGHT(_year, 2), "##") & " YTD</th>
            <th>" & _displayMonth3 & "</th>
            <th>" & _displayMonth2 & "</th>
            <th>" & _displayMonth1 & "</th>
            <th>" & _trending & "</th>
          </tr>"
        & _operationalRows
        & "
        </table>
      </td>
      <td class='right commentbackground'>
        <div class='commentsheader'></div>
        <div class='commentary'></div>
      </td>
    </tr>
    "
