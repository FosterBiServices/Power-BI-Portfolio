/* ============================================================
   METRIC LAYOUT TABLE
   ============================================================ */

WorkCare Operational Metric Layout =
DATATABLE(
    "SortOrder", INTEGER,
    "MetricName", STRING,
    "MetricKey", STRING,
{
    {1, "Client Revenue Attrition (<3%)", "_client"},
    {2, "Occ Health Encounter Volume to Target", "_encounter"},
    {3, "Call Answer SLA to Target", "_callSLA"},
    {4, "Occ Health Turnaround Time (<9 days)", "_turnaround"},
    {5, "Self-Care Rates to Target", "_retention_selfcare"}
}
)

/* ============================================================
   TREND COLUMN VARIABLE
   ============================================================ */

VAR _trending = " "

/* ============================================================
   TIME PLACEHOLDERS
   ============================================================ */

VAR _year = [Current Fiscal Year]

VAR _priorYear = [Prior Fiscal YTD]

VAR _currentmonthsort = [Current Month Sort]

VAR _month3 = [Month Year Prior 2]

VAR _month2 = [Month Year Prior 1]

VAR _month1 = [Month Year Current]

VAR _displayMonth3 = [Month Display Prior 2]

VAR _displayMonth2 = [Month Display Prior 1]

VAR _displayMonth1 = [Month Display Current]

/* ============================================================
   OPERATIONAL EXCELLENCE PLACEHOLDERS
   ============================================================ */

VAR __client = 1 // Client Revenue Attrition (<3%)
VAR __clientMonth3 = CALCULATE(__client,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __clientMonth2 = CALCULATE(__client,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __clientMonth = CALCULATE(__client,ALL(dimdate),dimdate[Month_Year] = _month1)
VAR __clientYtd = CALCULATE(__client,ALL(dimdate),dimdate[Fiscal Year] = _year,dimdate[Month Sort] <= _currentmonthsort)

VAR __encounter = 1 // Occ Health Encounter Volume to Target
VAR __encounterMonth3 = CALCULATE(__encounter,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __encounterMonth2 = CALCULATE(__encounter,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __encounterMonth = CALCULATE(__encounter,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __callSLA = 1 // Call Answer SLA to Target
VAR __callSLAMonth3 = CALCULATE(__callSLA,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __callSLAMonth2 = CALCULATE(__callSLA,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __callSLAMonth = CALCULATE(__callSLA,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __turnaround = 1 // Occ Health Turnaround Time (<9 days)
VAR __turnaroundMonth3 = CALCULATE(__turnaround,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __turnaroundMonth2 = CALCULATE(__turnaround,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __turnaroundMonth = CALCULATE(__turnaround,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __retention_selfcare = 1 // Self-Care Rates to Target
VAR __retention_selfcareMonth3 = CALCULATE(__retention_selfcare,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __retention_selfcareMonth2 = CALCULATE(__retention_selfcare,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __retention_selfcareMonth = CALCULATE(__retention_selfcare,ALL(dimdate),dimdate[Month_Year] = _month1)

/* ============================================================
   OPERATIONAL EXCELLENCE STATUS
   ============================================================ */

VAR __clientStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__clientMonth), BLANK(),
        __clientMonth >= 0.1, "red",
        __clientMonth >= 0.05, "amber",
        "green"
    )

VAR __encounterStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__encounterMonth), BLANK(),
        __encounterMonth >= 0.15, "red",
        __encounterMonth >= 0.1, "amber",
        "green"
    )

VAR __callSLAStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__callSLAMonth), BLANK(),
        __callSLAMonth >= 0.2, "red",
        __callSLAMonth >= 0.15, "amber",
        "green"
    )

VAR __turnaroundStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__turnaroundMonth), BLANK(),
        __turnaroundMonth >= 100, "red",
        __turnaroundMonth >= 25, "amber",
        "green"
    )

VAR __retention_selfcareStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__retention_selfcareMonth), BLANK(),
        __retention_selfcareMonth >= 0.3, "red",
        __retention_selfcareMonth >= 0.25, "amber",
        "green"
    )

/* ============================================================
   DYNAMIC HTML CREATION
   ============================================================ */
VAR _operationalRows =
    CONCATENATEX(
        'WorkCare Operational Metric Layout',
        VAR _metricKey = [MetricKey]

    VAR _status =
        SWITCH(
            _metricKey,
            "_client", __clientStatus,
            "_encounter", __encounterStatus,
            "_callSLA", __callSLAStatus,
            "_turnaround", __turnaroundStatus,
            "_retention_selfcare", __retention_selfcareStatus
        )

    VAR _ytd =
        SWITCH(
            _metricKey,
            "_client", __clientYtd
        )

    VAR _monthPrior2 =
        SWITCH(
            _metricKey,
            "_client", __clientMonth3,
            "_encounter", __encounterMonth3,
            "_callSLA", __callSLAMonth3,
            "_turnaround", __turnaroundMonth3,
            "_retention_selfcare", __retention_selfcareMonth3
        )

    VAR _monthPrior1 =
        SWITCH(
            _metricKey,
            "_client", __clientMonth2,
            "_encounter", __encounterMonth2,
            "_callSLA", __callSLAMonth2,
            "_turnaround", __turnaroundMonth2,
            "_retention_selfcare", __retention_selfcareMonth2
        )

    VAR _currentMonth =
        SWITCH(
            _metricKey,
            "_client", __clientMonth,
            "_encounter", __encounterMonth,
            "_callSLA", __callSLAMonth,
            "_turnaround", __turnaroundMonth,
            "_retention_selfcare", __retention_selfcareMonth
        )





        VAR _trendHtml = ""


    VAR _formatString =
        SWITCH(
            _metricKey,
            "_client", "0.0%",
            "_encounter", "0.0%",
            "_callSLA", "0.0%",
            "_turnaround", "#,##0",
            "_retention_selfcare", "0.0%"
        )

        RETURN
            "<tr>" &
            "<td class=label>" & [MetricName] & "</td>" &
            "<td class='status'>" &
                "<span class='dot " &
                    COALESCE(_status, "") &
                "'></span>" &
            "</td>" &
            "<td class='number'>" &
                IF(
                    ISBLANK(_ytd),
                    "",
                    FORMAT(_ytd, _formatString)
                ) &
                "</td>"
                & "<td class='number'>"
                & IF
                (
                    ISBLANK ( _monthPrior2 ),
                    "",
                    FORMAT ( _monthPrior2, _formatString )
                )
                & "</td>"
                & "<td class='number'>"
                & IF
                (
                    ISBLANK ( _monthPrior1 ),
                    "",
                    FORMAT ( _monthPrior1, _formatString )
                )
                & "</td>"
                & "<td class='number'>"
                & IF
                (
                    ISBLANK ( _currentMonth ),
                    "",
                    FORMAT ( _currentMonth, _formatString )
                )
                & "</td>" &
                "<td class='number'>"
                & IF
                (
                    ISBLANK ( _trendHtml ),
                    "",
                    _trendHtml
                )
                & "</td>" &
            "</td>" &
            "</tr>",
        "",
        [SortOrder],
        ASC
    )

RETURN
    "
    <!-- OPERATIONAL EXCELLENCE -->
    <tr>
      <td colspan='6' class='header'>
        Operational Excellence
      </td>
    </tr>
    <tr>
      <td class='left'>
        <table class='metric'>
          <colgroup>
            <col style='width:34%'>
            <col style='width:3%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
          </colgroup>

          <tr class='subheader'>
            <th></th>
            <th></th>
            <th>PY '" & FORMAT(RIGHT(_year,2),"##") & " YTD</th>
            <th>" & _displayMonth3 & "</th>
            <th>" & _displayMonth2 & "</th>
            <th>" & _displayMonth1 & "</th>
            <th>" & _trending & "</th>
          </tr>" &

    _operationalRows &

    "
        </table>
      </td>
      <td class='right commentbackground'>
        <div class='commentsheader'></div>
        <div class='commentary'></div>
      </td>
    </tr>
    "
