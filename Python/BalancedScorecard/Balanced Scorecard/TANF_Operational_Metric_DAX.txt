/* ============================================================
   METRIC LAYOUT TABLE
   ============================================================ */

TANF Operational Metric Layout =
DATATABLE(
    "SortOrder", INTEGER,
    "MetricName", STRING,
    "MetricKey", STRING,
{
    {1, "Retainage Accuracy %", "_retainage"},
    {2, "Referrals Generated", "_referrals"},
    {3, "Participants Served", "_participants"},
    {4, "Program Exits", "_exits"},
    {5, "Employment Rate Qtr of exit", "_empQtrofExit"},
    {6, "Employment Rate 1Q post-exit", "_emp1Q"},
    {7, "Employment Rate 2Q post-exit", "_emp2Q"},
    {8, "Median Earnings 2Q post-exit", "_earn2Q"},
    {9, "Median Earnings 4Q post-exit", "_earn4Q"},
    {10, "Level of Independence 4Q post-exit", "_independence"}
}
)

/* ============================================================
   TREND COLUMN VARIABLE
   ============================================================ */

VAR _trending = " "

/* ============================================================
   TIME PLACEHOLDERS
   ============================================================ */

VAR _year = [Current Fiscal Year]

VAR _priorYear = [Prior Fiscal YTD]

VAR _currentmonthsort = [Current Month Sort]

VAR _month3 = [Month Year Prior 2]

VAR _month2 = [Month Year Prior 1]

VAR _month1 = [Month Year Current]

VAR _displayMonth3 = [Month Display Prior 2]

VAR _displayMonth2 = [Month Display Prior 1]

VAR _displayMonth1 = [Month Display Current]

/* ============================================================
   OPERATIONAL EXCELLENCE PLACEHOLDERS
   ============================================================ */

VAR __retainage = 1 // Retainage Accuracy %
VAR __retainageMonth3 = CALCULATE(__retainage,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __retainageMonth2 = CALCULATE(__retainage,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __retainageMonth = CALCULATE(__retainage,ALL(dimdate),dimdate[Month_Year] = _month1)
VAR __retainageYtd = CALCULATE(__retainage,ALL(dimdate),dimdate[Fiscal Year] = _year,dimdate[Month Sort] <= _currentmonthsort)

VAR __referrals = 1 // Referrals Generated
VAR __referralsMonth3 = CALCULATE(__referrals,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __referralsMonth2 = CALCULATE(__referrals,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __referralsMonth = CALCULATE(__referrals,ALL(dimdate),dimdate[Month_Year] = _month1)
VAR __referralsYtd = CALCULATE(__referrals,ALL(dimdate),dimdate[Fiscal Year] = _year,dimdate[Month Sort] <= _currentmonthsort)
VAR __referralsPytd = CALCULATE(__referrals,ALL(dimdate),dimdate[Fiscal Year] = _priorYear,dimdate[Month Sort] <= _currentmonthsort)
VAR __referralsVar = __referralsYtd - __referralsPytd
VAR __referralsVariancePct = DIVIDE(__referralsVar,__referralsPytd)
VAR __referralsTrend = SWITCH( TRUE(), ISBLANK(__referrals), BLANK(), __referrals > 0, "up",  __referrals < 0, "down",  BLANK() )

VAR __participants = 1 // Participants Served
VAR __participantsMonth3 = CALCULATE(__participants,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __participantsMonth2 = CALCULATE(__participants,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __participantsMonth = CALCULATE(__participants,ALL(dimdate),dimdate[Month_Year] = _month1)
VAR __participantsYtd = CALCULATE(__participants,ALL(dimdate),dimdate[Fiscal Year] = _year,dimdate[Month Sort] <= _currentmonthsort)
VAR __participantsPytd = CALCULATE(__participants,ALL(dimdate),dimdate[Fiscal Year] = _priorYear,dimdate[Month Sort] <= _currentmonthsort)
VAR __participantsVar = __participantsYtd - __participantsPytd
VAR __participantsVariancePct = DIVIDE(__participantsVar,__participantsPytd)
VAR __participantsTrend = SWITCH( TRUE(), ISBLANK(__participants), BLANK(), __participants > 0, "up",  __participants < 0, "down",  BLANK() )

VAR __exits = 1 // Program Exits
VAR __exitsMonth3 = CALCULATE(__exits,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __exitsMonth2 = CALCULATE(__exits,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __exitsMonth = CALCULATE(__exits,ALL(dimdate),dimdate[Month_Year] = _month1)
VAR __exitsYtd = CALCULATE(__exits,ALL(dimdate),dimdate[Fiscal Year] = _year,dimdate[Month Sort] <= _currentmonthsort)
VAR __exitsPytd = CALCULATE(__exits,ALL(dimdate),dimdate[Fiscal Year] = _priorYear,dimdate[Month Sort] <= _currentmonthsort)
VAR __exitsVar = __exitsYtd - __exitsPytd
VAR __exitsVariancePct = DIVIDE(__exitsVar,__exitsPytd)
VAR __exitsTrend = SWITCH( TRUE(), ISBLANK(__exits), BLANK(), __exits > 0, "up",  __exits < 0, "down",  BLANK() )

VAR __empQtrofExit = 1 // Employment Rate Qtr of exit
VAR __empQtrofExitMonth3 = CALCULATE(__empQtrofExit,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __empQtrofExitMonth2 = CALCULATE(__empQtrofExit,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __empQtrofExitMonth = CALCULATE(__empQtrofExit,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __emp1Q = 1 // Employment Rate 1Q post-exit
VAR __emp1QMonth3 = CALCULATE(__emp1Q,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __emp1QMonth2 = CALCULATE(__emp1Q,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __emp1QMonth = CALCULATE(__emp1Q,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __emp2Q = 1 // Employment Rate 2Q post-exit
VAR __emp2QMonth3 = CALCULATE(__emp2Q,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __emp2QMonth2 = CALCULATE(__emp2Q,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __emp2QMonth = CALCULATE(__emp2Q,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __earn2Q = 1 // Median Earnings 2Q post-exit
VAR __earn2QMonth3 = CALCULATE(__earn2Q,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __earn2QMonth2 = CALCULATE(__earn2Q,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __earn2QMonth = CALCULATE(__earn2Q,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __earn4Q = 1 // Median Earnings 4Q post-exit
VAR __earn4QMonth3 = CALCULATE(__earn4Q,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __earn4QMonth2 = CALCULATE(__earn4Q,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __earn4QMonth = CALCULATE(__earn4Q,ALL(dimdate),dimdate[Month_Year] = _month1)

VAR __independence = 1 // Level of Independence 4Q post-exit
VAR __independenceMonth3 = CALCULATE(__independence,ALL(dimdate),dimdate[Month_Year] = _month3)
VAR __independenceMonth2 = CALCULATE(__independence,ALL(dimdate),dimdate[Month_Year] = _month2)
VAR __independenceMonth = CALCULATE(__independence,ALL(dimdate),dimdate[Month_Year] = _month1)

/* ============================================================
   OPERATIONAL EXCELLENCE STATUS
   ============================================================ */

VAR __retainageStatus =
    SWITCH(
        TRUE(),
        ISBLANK(__retainageMonth), BLANK(),
        __retainageMonth >= 45, "red",
        __retainageMonth >= 15, "amber",
        "green"
    )

VAR __referralsStatus = BLANK()

VAR __participantsStatus = BLANK()

VAR __exitsStatus = BLANK()

VAR __empQtrofExitStatus = BLANK()

VAR __emp1QStatus = BLANK()

VAR __emp2QStatus = BLANK()

VAR __earn2QStatus = BLANK()

VAR __earn4QStatus = BLANK()

VAR __independenceStatus = BLANK()

/* ============================================================
   DYNAMIC HTML CREATION
   ============================================================ */
VAR _operationalRows =
    CONCATENATEX(
        'TANF Operational Metric Layout',
        VAR _metricKey = [MetricKey]

    VAR _status =
        SWITCH(
            _metricKey,
            "_retainage", __retainageStatus,
            "_referrals", __referralsStatus,
            "_participants", __participantsStatus,
            "_exits", __exitsStatus,
            "_empQtrofExit", __empQtrofExitStatus,
            "_emp1Q", __emp1QStatus,
            "_emp2Q", __emp2QStatus,
            "_earn2Q", __earn2QStatus,
            "_earn4Q", __earn4QStatus,
            "_independence", __independenceStatus
        )

    VAR _ytd =
        SWITCH(
            _metricKey,
            "_retainage", __retainageYtd,
            "_referrals", __referralsYtd,
            "_participants", __participantsYtd,
            "_exits", __exitsYtd
        )

    VAR _monthPrior2 =
        SWITCH(
            _metricKey,
            "_retainage", __retainageMonth3,
            "_referrals", __referralsMonth3,
            "_participants", __participantsMonth3,
            "_exits", __exitsMonth3,
            "_empQtrofExit", __empQtrofExitMonth3,
            "_emp1Q", __emp1QMonth3,
            "_emp2Q", __emp2QMonth3,
            "_earn2Q", __earn2QMonth3,
            "_earn4Q", __earn4QMonth3,
            "_independence", __independenceMonth3
        )

    VAR _monthPrior1 =
        SWITCH(
            _metricKey,
            "_retainage", __retainageMonth2,
            "_referrals", __referralsMonth2,
            "_participants", __participantsMonth2,
            "_exits", __exitsMonth2,
            "_empQtrofExit", __empQtrofExitMonth2,
            "_emp1Q", __emp1QMonth2,
            "_emp2Q", __emp2QMonth2,
            "_earn2Q", __earn2QMonth2,
            "_earn4Q", __earn4QMonth2,
            "_independence", __independenceMonth2
        )

    VAR _currentMonth =
        SWITCH(
            _metricKey,
            "_retainage", __retainageMonth,
            "_referrals", __referralsMonth,
            "_participants", __participantsMonth,
            "_exits", __exitsMonth,
            "_empQtrofExit", __empQtrofExitMonth,
            "_emp1Q", __emp1QMonth,
            "_emp2Q", __emp2QMonth,
            "_earn2Q", __earn2QMonth,
            "_earn4Q", __earn4QMonth,
            "_independence", __independenceMonth
        )

    VAR _trendIcon =
        SWITCH(
            _metricKey,
            "_referrals", __referralsTrend,
            "_participants", __participantsTrend,
            "_exits", __exitsTrend
        )

    VAR _pyVAR =
        SWITCH(
            _metricKey,
            "_referrals", __referralsVariancePct,
            "_participants", __participantsVariancePct,
            "_exits", __exitsVariancePct
        )

        VAR _trendHtml =
            SWITCH(
                TRUE(),
                ISBLANK(_pyVAR), "",
                _trendIcon = "up",
                    "<span class='trend up'>▲ "
                        & FORMAT(_pyVAR, "0.0%")
                        & "</span>",
                _trendIcon = "down",
                    "<span class='trend down'>▼ "
                        & FORMAT(ABS(_pyVAR), "0.0%")
                        & "</span>",
                FORMAT(_pyVAR, "0.0%")
            )


    VAR _formatString =
        SWITCH(
            _metricKey,
            "_retainage", "0.0%",
            "_referrals", "#,##0",
            "_participants", "#,##0",
            "_exits", "#,##0",
            "_empQtrofExit", "#,##0",
            "_emp1Q", "#,##0",
            "_emp2Q", "#,##0",
            "_earn2Q", "#,##0",
            "_earn4Q", "#,##0",
            "_independence", "#,##0"
        )

        RETURN
            "<tr>" &
            "<td class=label>" & [MetricName] & "</td>" &
            "<td class='status'>" &
                "<span class='dot " &
                    COALESCE(_status, "") &
                "'></span>" &
            "</td>" &
            "<td class='number'>" &
                IF(
                    ISBLANK(_ytd),
                    "",
                    FORMAT(_ytd, _formatString)
                ) &
                "</td>"
                & "<td class='number'>"
                & IF
                (
                    ISBLANK ( _monthPrior2 ),
                    "",
                    FORMAT ( _monthPrior2, _formatString )
                )
                & "</td>"
                & "<td class='number'>"
                & IF
                (
                    ISBLANK ( _monthPrior1 ),
                    "",
                    FORMAT ( _monthPrior1, _formatString )
                )
                & "</td>"
                & "<td class='number'>"
                & IF
                (
                    ISBLANK ( _currentMonth ),
                    "",
                    FORMAT ( _currentMonth, _formatString )
                )
                & "</td>" &
                "<td class='number'>"
                & IF
                (
                    ISBLANK ( _trendHtml ),
                    "",
                    _trendHtml
                )
                & "</td>" &
            "</td>" &
            "</tr>",
        "",
        [SortOrder],
        ASC
    )

RETURN
    "
    <!-- OPERATIONAL EXCELLENCE -->
    <tr>
      <td colspan='6' class='header'>
        Operational Excellence
      </td>
    </tr>
    <tr>
      <td class='left'>
        <table class='metric'>
          <colgroup>
            <col style='width:34%'>
            <col style='width:3%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
            <col style='width:12.6%'>
          </colgroup>

          <tr class='subheader'>
            <th></th>
            <th></th>
            <th>PY '" & FORMAT(RIGHT(_year,2),"##") & " YTD</th>
            <th>" & _displayMonth3 & "</th>
            <th>" & _displayMonth2 & "</th>
            <th>" & _displayMonth1 & "</th>
            <th>" & _trending & "</th>
          </tr>" &

    _operationalRows &

    "
        </table>
      </td>
      <td class='right commentbackground'>
        <div class='commentsheader'></div>
        <div class='commentary'></div>
      </td>
    </tr>
    "
