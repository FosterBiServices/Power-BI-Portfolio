let
    // rows stored in the query as Base64 text of raw-deflate compressed JSON, so no data source is needed
    Source = Json.Document(
        Binary.Decompress(Binary.FromText("<base64>", BinaryEncoding.Base64), Compression.Deflate)
    ),
    Rows = Table.FromRows(
        Source,
        {
            "Date",
            "Scenario",
            "LocationKey",
            "ProductKey",
            "CustomerKey",
            "Units",
            "Gross Sales",
            "Discounts",
            "Net Sales",
            "Cost of Sales"
        }
    ),
    Typed = fxApplyTypes(
        Rows,
        {
            {"Date", type date},
            {"Scenario", type text},
            {"LocationKey", Int64.Type},
            {"ProductKey", Int64.Type},
            {"CustomerKey", Int64.Type},
            {"Units", Int64.Type},
            {"Gross Sales", Int64.Type},
            {"Discounts", Int64.Type},
            {"Net Sales", Int64.Type},
            {"Cost of Sales", Int64.Type}
