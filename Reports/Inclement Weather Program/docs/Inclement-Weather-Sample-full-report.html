let
  NormalizeTime = (Input as any) as nullable time =>
    let
      // Valid time = keep it
      _DirectTime =
        try Time.From(Input) otherwise null,

      Result =
        if _DirectTime <> null then
          _DirectTime
        else
          let
            // Normalize Text
            _Text   = try Text.From(Input) otherwise null,
            _Trim   = if _Text <> null then Text.Trim(_Text) else null,
            _Upper  = if _Trim <> null then Text.Upper(_Trim) else null,

            // Remove problematic characters
            _CleanChars =
              if _Upper <> null then
                Text.Select(_Upper, {"0".."9", ":", "A", "P", "M", " "})
              else null,

            // Fix spacing
            _FixSpacing =
              if _CleanChars <> null then
                Text.Replace(Text.Replace(_CleanChars, "AM", " AM"), "PM", " PM")
              else null,

            _Normalized =
              if _FixSpacing <> null then Text.Trim(_FixSpacing) else null,

            // ✅ KEY FIX: Handle simple numbers explicitly
            _AddMinutes =
              if _Normalized <> null and _Normalized <> "" and not Text.Contains(_Normalized, ":")
              then _Normalized & ":00"
              else _Normalized,

            // ✅ FINAL FIX: Convert the correct value
            _Time =
              try Time.From(_AddMinutes) otherwise null
          in
            _Time
    in
      Result,

  Transform =
    Table.TransformColumns(
      tbl,
      List.Transform(
        Columns,
        each { _, NormalizeTime, type time }
      )
    )

  Transform
